```dart
import 'dart:convert';
import 'package:http/http.dart' as http;

/// Citation Data Model
class Citation {
  final int sourceId;
  final String fileName;
  final int pageNumber;
  final String section;
  final String snippet;

  Citation({
    required this.sourceId,
    required this.fileName,
    required this.pageNumber,
    required this.section,
    required this.snippet,
  });
}

/// RAG Response Data Model
class RAGResponse {
  final String answer;
  final List<Citation> citations;
  final double confidenceScore;
  final bool isGrounded;

  RAGResponse({
    required this.answer,
    required this.citations,
    required this.confidenceScore,
    required this.isGrounded,
  });
}

/// Main MeghaRag Service for Flutter
class MeghaRagService {
  // Credentials
  static const String geminiApiKey = '<GEMINI_API_KEY>';
  static const String chromaApiKey = 'ck-45asi2qrJPZHUcf3EqmyfWiGSidHSo8cwAYa92ju7fKF';
  static const String tenantId = '5b24e72b-47d9-467c-b55e-288db51e3e55';
  static const String database = 'megha';
  static const String collectionName = 'megharag_docs';

  String? _cachedCollectionId;

  /// Fetch and cache Chroma Cloud collection UUID
  Future<String> _getCollectionId() async {
    if (_cachedCollectionId != null) return _cachedCollectionId!;

    final url = Uri.parse(
      'https://api.trychroma.com/api/v1/tenants/$tenantId/databases/$database/collections/$collectionName',
    );

    final response = await http.get(
      url,
      headers: {
        'Content-Type': 'application/json',
        'x-chroma-token': chromaApiKey,
      },
    );

    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      _cachedCollectionId = data['id'];
      return _cachedCollectionId!;
    } else {
      throw Exception('Failed to get Chroma Cloud collection ID: ${response.body}');
    }
  }

  /// 1. Generate query embedding vector using Gemini API
  Future<List<double>> embedQuery(String text) async {
    final url = Uri.parse(
      'https://generativelanguage.googleapis.com/v1beta/models/gemini-embedding-001:embedContent?key=$geminiApiKey',
    );

    final response = await http.post(
      url,
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({
        'model': 'models/gemini-embedding-001',
        'content': {
          'parts': [
            {'text': text}
          ]
        }
      }),
    );

    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      final List<dynamic> values = data['embedding']['values'];
      return values.map((v) => (v as num).toDouble()).toList();
    } else {
      throw Exception('Failed to embed query: ${response.body}');
    }
  }

  /// 2. Query Chroma Cloud REST API for top document chunks
  Future<List<Map<String, dynamic>>> searchVectorStore(List<double> vector, {int topK = 5}) async {
    final collectionId = await _getCollectionId();
    final url = Uri.parse(
      'https://api.trychroma.com/api/v1/tenants/$tenantId/databases/$database/collections/$collectionId/query',
    );

    final response = await http.post(
      url,
      headers: {
        'Content-Type': 'application/json',
        'x-chroma-token': chromaApiKey,
      },
      body: jsonEncode({
        'query_embeddings': [vector],
        'n_results': 15,
        'include': ['documents', 'metadatas', 'distances'],
      }),
    );

    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      if (data['documents'] == null || (data['documents'][0] as List).isEmpty) {
        return [];
      }

      final documents = data['documents'][0];
      final metadatas = data['metadatas'][0];
      final distances = data['distances'][0];

      List<Map<String, dynamic>> candidateChunks = [];
      for (int i = 0; i < documents.length; i++) {
        double dist = (distances[i] as num).toDouble();
        double simScore = 1.0 / (1.0 + (dist < 0 ? 0 : dist));

        candidateChunks.add({
          'source_id': i + 1,
          'text': documents[i] as String,
          'metadata': metadatas[i] as Map<String, dynamic>,
          'similarity_score': simScore,
        });
      }

      // Take top K sorted by similarity
      candidateChunks.sort((a, b) => (b['similarity_score'] as double).compareTo(a['similarity_score'] as double));
      return candidateChunks.take(topK).toList();
    } else {
      throw Exception('Failed to query Chroma Cloud: ${response.body}');
    }
  }

  /// 3. Execute Complete Grounded RAG Pipeline
  Future<RAGResponse> query(String queryText) async {
    // A. Embed query
    final vector = await embedQuery(queryText);

    // B. Search Chroma Cloud
    final retrievedChunks = await searchVectorStore(vector, topK: 5);

    if (retrievedChunks.isEmpty) {
      return RAGResponse(
        answer: "I couldn't find sufficient information in the uploaded documents to answer your question.",
        citations: [],
        confidenceScore: 0.0,
        isGrounded: true,
      );
    }

    // C. Format context string
    StringBuffer contextBuffer = StringBuffer();
    List<Citation> allCitations = [];

    for (int i = 0; i < retrievedChunks.length; i++) {
      final chunk = retrievedChunks[i];
      final meta = chunk['metadata'];
      int sourceId = i + 1;

      contextBuffer.writeln('[SOURCE $sourceId] File: ${meta['file_name'] ?? 'Unknown'} (Page ${meta['page_number'] ?? 1})');
      contextBuffer.writeln(chunk['text']);
      contextBuffer.writeln();

      allCitations.add(Citation(
        sourceId: sourceId,
        fileName: meta['file_name'] ?? 'Unknown',
        pageNumber: meta['page_number'] ?? 1,
        section: meta['section'] ?? '',
        snippet: (chunk['text'] as String).length > 120
            ? '${(chunk['text'] as String).substring(0, 120)}...'
            : chunk['text'] as String,
      ));
    }

    // D. Call Gemini 3.6 Flash Generation API
    final geminiUrl = Uri.parse(
      'https://generativelanguage.googleapis.com/v1beta/models/gemini-3.6-flash:generateContent?key=$geminiApiKey',
    );

    final response = await http.post(
      geminiUrl,
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({
        'system_instruction': {
          'parts': [
            {
              'text':
                  'You are MeghaRag, an expert AI assistant providing strictly grounded answers based ONLY on the provided context.\n\nStrict Guidelines:\n1. Answer the query using ONLY facts directly stated in the provided context.\n2. If context lacks sufficient information, state: "I couldn\'t find sufficient information in the uploaded documents to answer your question."\n3. Do NOT extrapolate or assume facts.\n4. Whenever citing facts, include bracketed source numbers like [SOURCE X] where X matches the given context block ID.'
            }
          ]
        },
        'contents': [
          {
            'parts': [
              {
                'text': '--- RETRIEVED DOCUMENT CONTEXT ---\n$contextBuffer\n\n--- USER QUESTION ---\n$queryText\n\n--- GROUNDED ANSWER ---'
              }
            ]
          }
        ],
        'generationConfig': {
          'temperature': 0.1,
          'maxOutputTokens': 1000,
        }
      }),
    );

    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      final rawAnswer = data['candidates'][0]['content']['parts'][0]['text'] as String;

      // Smart Citation Filter: Only return sources explicitly cited in answer text
      RegExp exp = RegExp(r'\[(?:SOURCE\s*)?(\d+)\]', caseSensitive: false);
      Iterable<RegExpMatch> matches = exp.allMatches(rawAnswer);
      Set<int> referencedIds = matches.map((m) => int.parse(m.group(1)!)).toSet();

      List<Citation> finalCitations = allCitations.where((c) => referencedIds.contains(c.sourceId)).toList();

      return RAGResponse(
        answer: rawAnswer,
        citations: finalCitations.isEmpty ? allCitations : finalCitations,
        confidenceScore: 0.95,
        isGrounded: true,
      );
    } else {
      throw Exception('Failed to generate answer from Gemini: ${response.body}');
    }
  }
}
```

Place the following Chat UI widget in `lib/views/chat_screen.dart`:

```dart
import 'package:flutter/material.dart';
import '../services/megha_rag_service.dart';

class ChatScreen extends StatefulWidget {
  const ChatScreen({Key? key}) : super(key: key);

  @override
  State<ChatScreen> createState() => _ChatScreenState();
}

class _ChatScreenState extends State<ChatScreen> {
  final TextEditingController _controller = TextEditingController();
  final MeghaRagService _ragService = MeghaRagService();

  bool _isLoading = false;
  String _answer = '';
  List<Citation> _citations = [];

  void _submitQuery() async {
    final query = _controller.text.trim();
    if (query.isEmpty) return;

    setState(() {
      _isLoading = true;
      _answer = '';
      _citations = [];
    });

    try {
      final res = await _ragService.query(query);
      setState(() {
        _answer = res.answer;
        _citations = res.citations;
      });
    } catch (e) {
      setState(() {
        _answer = 'Error: $e';
      });
    } finally {
      setState(() {
        _isLoading = false;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('MeghaRag AI Assistant')),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          children: [
            Expanded(
              child: SingleChildScrollView(
                child: Column(
                  crossAxisAlignment: CrossAlignment.start,
                  children: [
                    if (_isLoading)
                      const Center(child: CircularProgressIndicator())
                    else if (_answer.isNotEmpty) ...[
                      Container(
                        padding: const EdgeInsets.all(16),
                        decoration: BoxDecoration(
                          color: Colors.grey.shade900,
                          borderRadius: BorderRadius.circular(12),
                        ),
                        child: Text(
                          _answer,
                          style: const TextStyle(fontSize: 16, color: Colors.white),
                        ),
                      ),
                      const SizedBox(height: 16),
                      if (_citations.isNotEmpty) ...[
                        const Text(
                          '📌 Sources & Citations:',
                          style: TextStyle(fontWeight: FontWeight.bold, fontSize: 16),
                        ),
                        const SizedBox(height: 8),
                        ..._citations.map(
                          (c) => Card(
                            child: ListTile(
                              leading: CircleAvatar(child: Text('${c.sourceId}')),
                              title: Text('${c.fileName} (Page ${c.pageNumber})'),
                              subtitle: Text(c.snippet),
                            ),
                          ),
                        ),
                      ],
                    ],
                  ],
                ),
              ),
            ),
            Row(
              children: [
                Expanded(
                  child: TextField(
                    controller: _controller,
                    decoration: const InputDecoration(
                      hintText: 'Ask a question about your documents...',
                      border: OutlineInputBorder(),
                    ),
                  ),
                ),
                const SizedBox(width: 8),
                IconButton(
                  icon: const Icon(Icons.send),
                  onPressed: _isLoading ? null : _submitQuery,
                ),
              ],
            ),
          ],
        ),
      ),
    );
  }
}
```
